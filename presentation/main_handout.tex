\begin{frame}{Preprocessing Pipeline}
\small
\begin{itemize}[<+->]
  \item Reproducible setup: fixed seed (\texttt{SEED=42}) for split and initialization.
  \item Split strategy: \textbf{80/10/10} $\Rightarrow$ train 614, val 76, test 78.
  \item Train loader is shuffled; validation/test loaders are deterministic.
  \item Z-score normalization computed on train set and reused on val/test:
\end{itemize}

\vspace{-0.2em}
\[
\mathbf{x}_{norm} = \frac{\mathbf{x} - \boldsymbol{\mu}_{train}}{\boldsymbol{\sigma}_{train} + \epsilon}
\]
\vspace{-0.7em}
\small
\begin{itemize}[<+->]
  \item Numerical stability: replace zero std by small $\epsilon$.
\end{itemize}
\end{frame}

\section{Model}
\begin{frame}{Why LSTM for This Task?}
\small
\begin{itemize}[<+->]
  \item The class depends on \textbf{trajectory dynamics}, not one isolated timestep.
  \item From UE5 course (\texttt{3\_RNN\_Transformers\_2025.pdf}): vanilla RNN training suffers from vanishing/exploding gradients.
  \item LSTM introduces a memory cell with gates to control information flow:
\end{itemize}
\vspace{-0.2em}
\[
i_t, f_t, o_t = \sigma(\cdot),\quad
\tilde{c}_t = \tanh(\cdot),\quad
c_t = f_t \odot c_{t-1} + i_t \odot \tilde{c}_t
\]
\vspace{-0.7em}
\small
\begin{itemize}[<+->]
  \item This is aligned with course motivation: preserve useful long-term information.
\end{itemize}
\end{frame}

\begin{frame}{Final Architecture and Training Setup}
\small
\begin{itemize}[<+->]
  \item Input: sequence \(\texttt{(batch, 150, 20)}\)
  \item Backbone: LSTM (\texttt{num\_layers=2}, \texttt{hidden\_size=64}, \texttt{dropout=0.2})
  \item Sequence-to-class head: last hidden state \(\rightarrow\) \texttt{Linear(64, 3)}
  \item Loss: \texttt{CrossEntropyLoss}
  \item Optimizer: \texttt{Adam(lr=1e-3)}
  \item Batch size: \texttt{32}, training budget: up to \texttt{40} epochs
\end{itemize}
\end{frame}

\section{Challenges Encountered}
\begin{frame}{Practical Issues During Development}
\footnotesize
\begin{itemize}[<+->]
  \item \textbf{Reproducibility}: without fixed seeds, run-to-run comparison was unreliable.
  \item \textbf{Optimization sensitivity}: preprocessing and regularization strongly changed convergence behavior.
\end{itemize}

\begin{frame}{Exploding Gradients and Fix}
\begin{columns}[t,onlytextwidth]
  \column{0.52\textwidth}
  \includegraphics[width=0.96\linewidth]{gradient_explosion.png}
  {\scriptsize Observed spikes in gradient norm across training steps.}
  \column{0.44\textwidth}
  \footnotesize
  \textbf{Problem}\\
  Large gradient spikes destabilized updates.
  \vspace{0.4em}

  \textbf{Mitigation}\\
  Gradient clipping in the training loop with max norm \(1.0\):\\
  PyTorch \textit{clip\_grad\_norm\_}(\ldots).
\end{columns}
\end{frame}

\begin{frame}{Conclusion}
